RewriteEngine On # Redirect /index.html to /index RewriteCond %{THE_REQUEST} \s/+(.*/)?index\.html[\s?] [NC] RewriteRule ^(.*/)?index\.html$ /$1index [R=301,L] # Redirect any .html URL to extensionless URL RewriteCond %{THE_REQUEST} \s/+(.+)\.html[\s?] [NC] RewriteRule ^(.+)\.html$ /$1 [R=301,L] # Internally load the .html file RewriteCond %{REQUEST_FILENAME} !-d RewriteCond %{REQUEST_FILENAME}.html -f RewriteRule ^(.+?)/?$ $1.html [L]
